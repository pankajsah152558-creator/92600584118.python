(3)	Write a Python Program to enter Principle Amt., Rate of Interest and No. of Years and find Simple and Compound Interest.

# Program to calculate Simple Interest and Compound Interest

p = float(input("Enter Principal Amount: "))
r = float(input("Enter Rate of Interest: "))
n = int(input("Enter Number of Years: "))

# Simple Interest
SI = (p * r * n) / 100

# Compound Interest
CI = p * (1 + r / 100) ** n - p

print("Simple Interest =", SI)
print("Compound Interest =", CI)

(11)	Write a Python Program to enter a no. and find sum of its digits

# Program to find sum of digits of a number

# Input number
num = int(input("Enter a number: "))

# Initialize sum
digit_sum = 0

# Loop to extract digits
while num > 0:
    digit = num % 10      # Get last digit
    digit_sum += digit    # Add to sum
    num //= 10            # Remove last digit

# Display result
print("Sum of digits =", digit_sum)


(12)	Write a Python Program to enter a no. and find its reverse

# Program to find reverse of a number

# Input number
num = int(input("Enter a number: "))

# Initialize reverse
reverse_num = 0

# Loop to reverse digits
while num > 0:
    digit = num % 10              # Get last digit
    reverse_num = reverse_num * 10 + digit  # Build reversed number
    num //= 10                    # Remove last digit

# Display result
print("Reversed number =", reverse_num)


(13)	Write a Python Program to enter a no. and check it is palindrome or not?

# Program to check if a number is palindrome

# Input number
num = int(input("Enter a number: "))

# Store original number
original_num = num

# Reverse the number
reverse_num = 0
while num > 0:
    digit = num % 10
    reverse_num = reverse_num * 10 + digit
    num //= 10

# Check palindrome
if original_num == reverse_num:
    print(f"{original_num} is a Palindrome.")
else:
    print(f"{original_num} is NOT a Palindrome.")


(14)	Write a Python Program to enter a no. and check it is Armstrong or not?  

# Program to check if a number is Armstrong

# Input number
num = int(input("Enter a number: "))

# Store original number
original_num = num

# Count digits
digits = len(str(num))

# Calculate sum of powers of digits
sum_of_powers = 0
while num > 0:
    digit = num % 10
    sum_of_powers += digit ** digits
    num //= 10

# Check Armstrong condition
if original_num == sum_of_powers:
    print(f"{original_num} is an Armstrong number.")
else:
    print(f"{original_num} is NOT an Armstrong number.")


(15)	Write a Python Program to enter a no. and check it is prime or not?

# Program to check if a number is prime

# Input number
num = int(input("Enter a number: "))

# Prime check
if num <= 1:
    print(f"{num} is NOT a Prime number.")
else:
    is_prime = True
    for i in range(2, int(num**0.5) + 1):   # check divisors up to sqrt(num)
        if num % i == 0:
            is_prime = False
            break

    if is_prime:
        print(f"{num} is a Prime number.")
    else:
        print(f"{num} is NOT a Prime number.")

# Program to check if a number is KrishnaMurthy (Strong number)

# Input number
num = int(input("Enter a number: "))

# Store original number
original_num = num

# Function to calculate factorial of a digit
def factorial(n):
    fact = 1
    for i in range(1, n + 1):
        fact *= i
    return fact

# Calculate sum of factorials of digits
sum_of_factorials = 0
while num > 0:
    digit = num % 10
    sum_of_factorials += factorial(digit)
    num //= 10

# Check KrishnaMurthy condition
if original_num == sum_of_factorials:
    print(f"{original_num} is a KrishnaMurthy number.")
else:
    print(f"{original_num} is NOT a KrishnaMurthy number.")
